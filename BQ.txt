-----Daily_Demand by: Avg Q3 For year 2017, 2018
WITH StoreProductMetrics AS (
SELECT
    Store_ID, Product_ID
, SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2018 THEN Units ELSE 0 END) / 92.0 AS Avg_2018
, SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2017 THEN Units ELSE 0 END) / 92.0 AS Avg_2017
FROM toys.sales
WHERE EXTRACT(QUARTER FROM Date) = 3
    GROUP BY Store_ID, Product_ID
)
SELECT spm.Store_ID, inv.product_id, inv.stock_on, Avg_2018 AS daily_demand_2018
--calc inventory VS daily_demand
, CASE
    WHEN (spm.Avg_2018 / COALESCE(spm.Avg_2017,0)) >1
    THEN (spm.Avg_2018 / COALESCE(spm.Avg_2017,0))
ELSE 1
END AS stock_balance
FROM StoreProductMetrics spm JOIN toys.inventory inv ON spm.Store_ID=inv.store_id

WITH StoreProductMetrics AS (
    SELECT
        Store_ID,
        Product_ID,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2018 THEN Units ELSE 0 END) / 92.0 AS Avg_2018,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2017 THEN Units ELSE 0 END) / 92.0 AS Avg_2017
    FROM `toys.sales`
    WHERE EXTRACT(QUARTER FROM Date) = 3
    GROUP BY Store_ID, Product_ID
)
SELECT
    spm.Store_ID,
    inv.Product_ID,
    inv.Stock_On, -- תיקון שם העמודה מ-Stock_On_Hand ל-Stock_On
    spm.Avg_2018 AS daily_demand_2018,
    -- חישוב יחס צמיחה עם הגנה מחלוקה ב-0
    CASE
        WHEN (spm.Avg_2018 / NULLIF(spm.Avg_2017, 0)) > 1
        THEN (spm.Avg_2018 / NULLIF(spm.Avg_2017, 0))
        ELSE 1
    END AS growth_factor,
    -- כמה ימים המלאי יספיק (מלאי חלקי דרישה)
    ROUND(inv.Stock_On / NULLIF(spm.Avg_2018, 0), 2) AS days_of_stock
FROM StoreProductMetrics spm
JOIN `toys.inventory` inv ON spm.Store_ID = inv.Store_ID AND spm.Product_ID = inv.Product_ID
ORDER BY days_of_stock ASC;

WITH StoreProductMetrics AS (
    SELECT
        Store_ID,
        Product_ID,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2018 THEN Units ELSE 0 END) / 92.0 AS Avg_2018,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2017 THEN Units ELSE 0 END) / 92.0 AS Avg_2017
    FROM `toys.sales`
    WHERE EXTRACT(QUARTER FROM Date) = 3
    GROUP BY Store_ID, Product_ID
)
SELECT
    s.Store_Location,
    spm.Store_ID,
    inv.Product_ID,
    inv.Stock_On,
    spm.Avg_2017 AS daily_demand_2017,
    spm.Avg_2018 AS daily_demand_2018,
    -- כאן אנחנו משתמשים ב-IFNULL כדי שמוצרים ללא דרישה יקבלו ערך גבוה (כמו 999) ולא NULL
    IFNULL(ROUND(inv.Stock_On / NULLIF(spm.Avg_2018, 0), 2), 999) AS days_of_stock
FROM StoreProductMetrics spm
JOIN `toys.inventory` inv ON spm.Store_ID = inv.Store_ID AND spm.Product_ID = inv.Product_ID
JOIN toys.stores s on inv.store_id = s.Store_ID
WHERE s.Store_Location = 'Downtown'
ORDER BY days_of_stock ASC;

WITH StoreProductMetrics AS (
    SELECT
        CAST(Store_ID AS INT64) AS Store_ID,
        CAST(Product_ID AS INT64) AS Product_ID,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2018 THEN Units ELSE 0 END) / 92.0 AS Avg_2018,
        SUM(CASE WHEN EXTRACT(YEAR FROM Date) = 2017 THEN Units ELSE 0 END) / 92.0 AS Avg_2017
    FROM `toys.sales`
    WHERE EXTRACT(QUARTER FROM Date) = 3
    GROUP BY 1, 2
)
SELECT
  ---  s.Store_Location,
    spm.Store_ID,
    spm.Product_ID,
    inv.Stock_On,
    spm.Avg_2017 AS daily_demand_2017,
    spm.Avg_2018 AS daily_demand_2018,
    -- חישוב ימי המלאי עם טיפול בערכי NULL ו-0
    IFNULL(ROUND inv.Stock_On) / NULLIF(spm.Avg_2018, 0), 2), 999) AS days_of_stock
FROM StoreProductMetrics spm
-- LEFT JOIN מבטיח שאם יש מכירות, השורה לא תימחק גם אם אין נתוני מלאי
LEFT JOIN `toys.inventory` inv
    ON spm.Store_ID = CAST(inv.Store_ID AS INT64)
    AND spm.Product_ID = CAST(inv.Product_ID AS INT64)
LEFT JOIN `toys.stores` s
    ON spm.Store_ID = CAST(s.Store_ID AS INT64)
---WHERE UPPER(s.Store_Location) = 'DOWNTOWN'
ORDER BY days_of_stock ASC;
